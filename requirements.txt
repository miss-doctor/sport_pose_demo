# 本机实测通过的版本组合（Windows 11 + CPU + Python 3.11）
#
# 顺序很重要：mmcv 必须先于 mmdet / mmpose 安装，且必须从 OpenMMLab
# 官方索引取对应 torch / Python / 平台的预编译轮子。
# 直接用 scripts/ 下的脚本安装即可，不要 pip install -r 本文件。

# ---- 基础 ----
numpy<2                 # xtcocotools 的 C 扩展按 numpy 1.x 编译，2.x 会崩
setuptools<81           # setuptools 81+ 移除了 mmengine 依赖的 pkg_resources
opencv-python
matplotlib

# ---- torch（CPU 版；有 GPU 时改成自己的 CUDA 版本）----
torch==2.1.2+cpu
torchvision==0.16.2+cpu

# ---- OpenMMLab（mmcv 需来自官方索引，PyPI 上没有 Windows 轮子）----
mmcv==2.1.0
mmengine==0.10.7
mmdet==3.3.0
mmpose==1.3.2

# ---- 导出 / 校验（可选）----
# 注意：不要把输出目录命名为 onnx/，否则会遮蔽 onnx 包导致导出失败
onnx==1.15.0
onnxruntime==1.16.3
